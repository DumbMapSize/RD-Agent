# MODEL_TYPE = "Tabular"
# BATCH_SIZE = 32
# NUM_FEATURES = 10
# NUM_TIMESTEPS = 4
# NUM_EDGES = 20
# INPUT_VALUE = 1.0
# PARAM_INIT_VALUE = 1.0

import pickle

import torch
from model import model_cls

if MODEL_TYPE == "Tabular":
    input_shape = (BATCH_SIZE, NUM_FEATURES)
    m = model_cls(num_features=input_shape[1])
    data = torch.full(input_shape, INPUT_VALUE)
elif MODEL_TYPE == "TimeSeries":
    input_shape = (BATCH_SIZE, NUM_TIMESTEPS, NUM_FEATURES)
    m = model_cls(num_features=input_shape[2], num_timesteps=input_shape[1])
    data = torch.full(input_shape, INPUT_VALUE)
elif MODEL_TYPE == "Graph":
    node_feature = torch.randn(BATCH_SIZE, NUM_FEATURES)
    edge_index = torch.randint(0, BATCH_SIZE, (2, NUM_EDGES))
    m = model_cls(num_features=NUM_FEATURES)
    data = (node_feature, edge_index)
else:
    raise ValueError(f"Unsupported model type: {MODEL_TYPE}")

required_feature_names = getattr(m, "required_feature_names", ())
if isinstance(required_feature_names, str):
    required_feature_names = (required_feature_names,)
elif not isinstance(required_feature_names, (list, tuple)):
    raise ValueError("required_feature_names must be a string, list, or tuple")
required_feature_names = tuple(str(name) for name in required_feature_names)
if len(set(required_feature_names)) != len(required_feature_names):
    raise ValueError("required_feature_names contains duplicates")
if len(required_feature_names) > NUM_FEATURES:
    raise ValueError("required_feature_names exceeds the simulated feature count")

feature_names = list(required_feature_names)
feature_number = 0
while len(feature_names) < NUM_FEATURES:
    candidate = f"FEATURE_{feature_number}"
    feature_number += 1
    if candidate not in feature_names:
        feature_names.append(candidate)
feature_names = tuple(feature_names)
m.feature_names = feature_names
m.feature_index = {name: position for position, name in enumerate(feature_names)}

# Initialize all parameters of `m` to `param_init_value`
for _, param in m.named_parameters():
    param.data.fill_(PARAM_INIT_VALUE)

# Execute the model
if MODEL_TYPE == "Graph":
    out = m(*data)
else:
    out = m(data)

execution_model_output = out.cpu().detach().numpy()
execution_feedback_str = f"Execution successful, output tensor shape: {execution_model_output.shape}"

pickle.dump(execution_model_output, open("execution_model_output.pkl", "wb"))
pickle.dump(execution_feedback_str, open("execution_feedback_str.pkl", "wb"))
